function throttle(fn, wait) {
    if (typeof fn !== 'function')
        throw new TypeError('Expected a function');
    if (!Number.isFinite(wait) || wait < 0 || wait > 2147483647)
        throw new RangeError('wait must fit the timer range');
    let timer = null;
    let latestArgs = null;
    let latestReceiver;
    function invoke() {
        const args = latestArgs;
        const receiver = latestReceiver;
        latestArgs = null;
        latestReceiver = undefined;
        if (args !== null)
            fn.apply(receiver, args);
    }
    function tick() {
        timer = null;
        if (latestArgs !== null) {
            timer = setTimeout(tick, wait);
            invoke();
        }
    }
    function throttled(...args) {
        latestArgs = args;
        latestReceiver = this;
        if (timer === null) {
            timer = setTimeout(tick, wait);
            invoke();
        }
    }
    throttled.cancel = function () {
        if (timer !== null)
            clearTimeout(timer);
        timer = null;
        latestArgs = null;
        latestReceiver = undefined;
    };
    return throttled;
}
