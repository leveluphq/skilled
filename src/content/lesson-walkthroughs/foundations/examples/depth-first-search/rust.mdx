fn dfs_order(graph: &[Vec<i32>], start: i32) -> Vec<i32> {
    if start < 0 || start as usize >= graph.len() { return Vec::new(); }
    let mut stack = vec![start];
    let mut visited = vec![false; graph.len()];
    let mut output = Vec::new();
    while let Some(node) = stack.pop() {
        if visited[node as usize] { continue; }
        visited[node as usize] = true;
        output.push(node);
        for &neighbor in graph[node as usize].iter().rev() {
            if !visited[neighbor as usize] { stack.push(neighbor); }
        }
    }
    output
}
