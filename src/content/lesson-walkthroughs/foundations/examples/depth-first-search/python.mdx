def dfs_order(graph: list[list[int]], start: int) -> list[int]:
    if start < 0 or start >= len(graph):
        return []
    stack, visited, output = [start], set(), []
    while stack:
        node = stack.pop()
        if node in visited:
            continue
        visited.add(node)
        output.append(node)
        for neighbor in reversed(graph[node]):
            if neighbor not in visited:
                stack.append(neighbor)
    return output
