import java.util.*;

class DfsOrderExample {
static List<Integer> dfsOrder(List<List<Integer>> graph, int start) {
    List<Integer> output = new ArrayList<>();
    if (start < 0 || start >= graph.size()) return output;
    Deque<Integer> stack = new ArrayDeque<>();
    Set<Integer> visited = new HashSet<>();
    stack.addLast(start);
    while (!stack.isEmpty()) {
        int node = stack.removeLast();
        if (!visited.add(node)) continue;
        output.add(node);
        List<Integer> neighbors = graph.get(node);
        for (int index = neighbors.size() - 1; index >= 0; index--) {
            int neighbor = neighbors.get(index);
            if (!visited.contains(neighbor)) stack.addLast(neighbor);
        }
    }
    return output;
}
}
