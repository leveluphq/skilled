```cpp
#include <vector>
#include <optional>
#include <functional>
#include <algorithm>

int maxDepth(const std::vector<std::optional<int>>& values) {
    std::function<int(std::size_t)> height = [&](std::size_t index) -> int {
        if (index >= values.size() || !values[index]) return -1;
        return 1 + std::max(height(2 * index + 1), height(2 * index + 2));
    };
    return height(0);
}
```
