fun fibMemoStats(n: Int): List<Int> {
    val memo = mutableMapOf<Int, Int>()
    fun fib(value: Int): Int {
        if (memo.containsKey(value)) return memo.getValue(value)
        val answer = if (value < 2) value else fib(value - 1) + fib(value - 2)
        memo[value] = answer
        return answer
    }
    val answer = fib(n)
    return listOf(answer, memo.size)
}
