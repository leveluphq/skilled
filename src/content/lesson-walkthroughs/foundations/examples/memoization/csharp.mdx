using System;
using System.Collections.Generic;

public static class FibMemoStatsExample {
public static List<int> FibMemoStats(int n) {
    var memo = new Dictionary<int, int>();
    int Fib(int value) {
        if (memo.TryGetValue(value, out int cached)) return cached;
        int answer = value < 2 ? value : Fib(value - 1) + Fib(value - 2);
        memo[value] = answer;
        return answer;
    }
    int answer = Fib(n);
    return new List<int> { answer, memo.Count };
}
}
