def fib_memo_stats(n: int) -> list[int]:
    memo = {}
    def fib(value):
        if value in memo:
            return memo[value]
        answer = value if value < 2 else fib(value - 1) + fib(value - 2)
        memo[value] = answer
        return answer
    answer = fib(n)
    return [answer, len(memo)]
