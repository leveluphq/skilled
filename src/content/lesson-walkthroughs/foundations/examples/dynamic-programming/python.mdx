def minimum_coins(coins: list[int], amount: int) -> int:
    impossible = amount + 1
    best = [impossible] * (amount + 1)
    best[0] = 0
    for value in range(1, amount + 1):
        for coin in coins:
            if coin <= value and best[value - coin] != impossible:
                best[value] = min(best[value], 1 + best[value - coin])
    return -1 if best[amount] == impossible else best[amount]
