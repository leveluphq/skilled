function inOrder(values: readonly (number | null)[]): number[] {
  const output: number[] = [];
  function walk(index: number): void {
    if (index >= values.length || values[index] === null) return;
    walk(2 * index + 1);
    output.push(values[index]!);
    walk(2 * index + 2);
  }
  walk(0);
  return output;
}
