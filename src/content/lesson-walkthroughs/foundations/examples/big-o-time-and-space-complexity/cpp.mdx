#include <vector>

std::vector<std::vector<int>> suffixCopies(const std::vector<int>& values) {
    std::vector<std::vector<int>> suffixes;
    for (std::size_t index = 0; index < values.size(); ++index) {
        suffixes.emplace_back(values.begin() + index, values.end());
    }
    return suffixes;
}
