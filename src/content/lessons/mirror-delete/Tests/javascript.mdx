class ListNode {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

const buildLinkedList = array => {
  const head = new ListNode(array[0]);
  let node = head;

  for (let i = 1; i < array.length; i++) {
    node.next = new ListNode(array[i]);
    node = node.next;
  }

  return head;
};

const listToArray = head => {
  const results = [];
  let node = head;

  while (node) {
    results.push(node.data);
    node = node.next;
  }

  return results;
};
