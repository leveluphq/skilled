# If immutable, do operations on an array and combine
def duplicate_string(str, num_duplicates):
    solution_array = []

    for i in range(num_duplicates):
        # .push is O(1)
        solution_array.append(str)

        # Faster than str += str which would be O(n) for copying each time

    return ''.join(solution_array)
