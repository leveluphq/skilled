// If immutable, do operations on a StringBuilder (a resizable array of characters) and combine
static String duplicateString(String str, int numDuplicates) {
    StringBuilder solution = new StringBuilder();

    for (int i = 0; i < numDuplicates; i++) {
        // .append only copies the new characters to the end, like an array push
        solution.append(str);

        // Faster than str += str which would be O(n) for copying each time
    }

    return solution.toString();
}
