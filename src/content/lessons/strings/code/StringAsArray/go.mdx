```go
str := "Hello, world!"
array := []string{"H", "e", "l", "l", "o", ",", " ", "w", "o", "r", "l", "d", "!"}

// Transform an array to a string and compare it to the original string
fmt.Println(str == strings.Join(array, ""))
// returns true

// []string{"H", "e", "l", "l", "o", ",", " ", "w", "o", "r", "l", "d", "!"}
```
