```javascript
const string = 'Hello, world!';
const array = ['H', 'e', 'l', 'l', 'o', ',', ' ', 'w', 'o', 'r', 'l', 'd', '!'];

// Transform an array to a string and compare it to the original string
console.log(string === array.join(''));
// returns true

console.log(string.split('')); // Transform a string to an array
// ['H','e','l','l','o',',',' ','w','o','r','l','d','!']
```
