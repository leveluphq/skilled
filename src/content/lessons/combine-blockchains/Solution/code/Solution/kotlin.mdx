class Solution {
    fun combineBlockchains(chainHead1: ListNode?, chainHead2: ListNode?): ListNode? {
        // If one of the chains is empty, the combined chain is the other one
        if (chainHead1 == null) {
            return chainHead2
        } else if (chainHead2 == null) {
            return chainHead1
        }

        // Store the head of the first item in the combined chain
        val resultHead = if (chainHead1.data < chainHead2.data) chainHead1 else chainHead2
        // Init the combined list with the first item
        var combinedTail: ListNode = resultHead
        var currentNode1: ListNode? = chainHead1
        var currentNode2: ListNode? = chainHead2

        // Move the node forward that was already added to our resultHead
        if (chainHead1.data < chainHead2.data) {
            currentNode1 = chainHead1.next
        } else {
            currentNode2 = chainHead2.next
        }

        // We iterate through until one of the linked lists has been depleted
        while (currentNode1 != null && currentNode2 != null) {
            // In either case, we do the same thing.
            // We attach the node to our combined list,
            // and then we move forward in the contributed chain's list
            val nextNode: ListNode
            if (currentNode1.data < currentNode2.data) {
                nextNode = currentNode1
                currentNode1 = currentNode1.next
            } else {
                nextNode = currentNode2
                currentNode2 = currentNode2.next
            }
            combinedTail.next = nextNode

            // Move to the new tail in the combined chain
            combinedTail = nextNode
        }

        // One of the chains has been depleted (reached null)
        // The other has its remaining items in sorted order,
        // so we can just set the next value of our
        // combined list to the non-null node
        combinedTail.next = currentNode1 ?: currentNode2

        // ^ NOTE: The elvis operator ?: yields currentNode1 if it isn't null,
        // otherwise it yields currentNode2

        return resultHead
    }
}
