TAIL_POINTER = None

def has_loop(head):
    # Both runners start at the first node
    fast = head
    slow = head

    # Walk through the nodes and only break if the fast runner reaches the tail
    while fast != TAIL_POINTER and fast.next != TAIL_POINTER:
        # Move our slow pointer 1 node and fast pointer 2 nodes
        slow = slow.next
        fast = fast.next.next

        # If the fast meets the slow pointer at the same node, we know we have a cycle.
        # We compare the nodes themselves (with `is`) and not their data, since different nodes can hold the same data.
        if slow is fast:
            return True

    # If our fast pointer above hits a None node,
    # we break the `while` loop and know we have reached the tail
    return False
