class ListNode {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

// Links the nodes in order, then points the tail back to the node at loopIndex
const buildLinkedList = (values, loopIndex) => {
  const nodes = values.map(value => new ListNode(value));

  for (let i = 0; i < nodes.length - 1; i++) {
    nodes[i].next = nodes[i + 1];
  }

  if (loopIndex !== null) {
    nodes[nodes.length - 1].next = nodes[loopIndex];
  }

  return nodes.length > 0 ? nodes[0] : null;
};

const validate = ({ input, expected, description }) => {
  const head = buildLinkedList(input.values, input.loopIndex);
  const userAnswer = hasLoop(head);
  const passed = userAnswer === expected;

  return { passed, description, expected, userAnswer };
};
