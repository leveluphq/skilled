    # Links the nodes in order, then points the tail back to the node at loop_index (None means there is no loop)
    def buildLinkedList(self, values, loop_index):
        nodes = [ListNode(value) for value in values]

        for i in range(len(nodes) - 1):
            nodes[i].next = nodes[i + 1]

        if loop_index != None:
            nodes[-1].next = nodes[loop_index]

        return nodes[0] if len(nodes) > 0 else None
