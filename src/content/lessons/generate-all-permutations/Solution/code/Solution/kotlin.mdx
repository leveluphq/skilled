class Solution {
    fun generatePermutations(s: String): List<String> {
        // ** Base case ** - if the string is empty or 1 character, we just return that value
        if (s.length < 2) {
            return listOf(s)
        }

        val permutations = mutableListOf<String>()

        // Iterate through all the characters in the string
        for (i in s.indices) {
            // We move a character to the front and recursively find the permutations of all the characters after it
            // Note that the "front" is relative to a given permutation depth
            val frontCharacter = s[i]

            // Remove the new front character from the string
            val remainingChars = s.substring(0, i) + s.substring(i + 1)

            // Recursive call to generate a list of the permutations of the remaining characters
            val remainingPermutations = generatePermutations(remainingChars)

            // Add each permutation to the result for a given depth
            for (permutation in remainingPermutations) {
                permutations.add(frontCharacter + permutation)
            }
        }

        return permutations
    }
}
