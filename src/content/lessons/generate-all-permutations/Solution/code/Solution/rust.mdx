impl Solution {
    pub fn generate_permutations(s: String) -> Vec<String> {
        let characters: Vec<char> = s.chars().collect();

        // ** Base case ** - if the string is empty or 1 character, we just return that value
        if characters.len() < 2 {
            return vec![s];
        }

        let mut permutations = Vec::new();

        // Iterate through all the characters in the string
        for i in 0..characters.len() {
            // We move a character to the front and recursively find the permutations of all the characters after it
            // Note that the "front" is relative to a given permutation depth
            let front_character = characters[i];

            // Remove the new front character from the string
            let remaining_chars: String =
                characters[..i].iter().chain(&characters[i + 1..]).collect();

            // Recursive call to generate the permutations of the remaining characters
            let remaining_permutations = Self::generate_permutations(remaining_chars);

            // Add each permutation to the result for a given depth
            for permutation in remaining_permutations {
                permutations.push(format!("{front_character}{permutation}"));
            }
        }

        permutations
    }
}
