const validate = ({ input, expected, description }) => {
  const userAnswer = generatePermutations(input);

  // An answer that isn't a list (such as a missing return) is a failed test, not a crash
  if (!Array.isArray(userAnswer)) {
    return { passed: false, description, expected, userAnswer };
  }

  let passed = userAnswer.length === expected.length;

  userAnswer.sort();
  expected.sort();

  if (passed) {
    for (let i = 0; i < userAnswer.length; i++) {
      if (userAnswer[i] !== expected[i]) {
        passed = false;
      }
    }
  }


  return { passed, description, expected, userAnswer };
};
