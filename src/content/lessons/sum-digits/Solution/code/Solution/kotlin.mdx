```kotlin
class Solution {
    fun sumDigits(n: Int): Int {
        // ** Base case ** - we know to stop when our number no longer has digits
        if (n == 0) {
            return 0
        }

        /*
          Recursive Case:
            1. Extract the digit in the one's place using % 10
            2. Remove the one's place by dividing by 10 (Int division drops the remainder)
            3. Sum it recursively with the rest of the digits in the integer
        */
        val onesPlaceValue = n % 10
        val trimmedItems = n / 10
        val sumOfRemainingItems = sumDigits(trimmedItems) // recursive call

        return onesPlaceValue + sumOfRemainingItems
    }
}
```
