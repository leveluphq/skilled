class Queue {
  queue: string[];

  constructor() {
    this.queue = [];
  }
  // Assume this actually operates in O(1) time
  // as an optimized queue should
  enqueue(item: string) {
    this.queue.unshift(item);
  }
  dequeue(): string {
    return this.queue.pop();
  }
  size(): number {
    return this.queue.length;
  }
