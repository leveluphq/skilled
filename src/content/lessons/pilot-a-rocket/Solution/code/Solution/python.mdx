from collections import deque

START_NODE = 'base'
END_NODE = 'ship'

def pilot_rocket(satellites):
    # Queue to handle BFS
    bfs_node_queue = Queue()
    bfs_node_queue.enqueue(START_NODE)

    # Use a hash table to track the node that came
    # prior to the current in our BFS to track the path
    previous_node_in_path = {}
    previous_node_in_path[START_NODE] = None

    # Perform our BFS
    while bfs_node_queue.size() > 0:
        node = bfs_node_queue.dequeue()

        # We've successfully reached the end, and now we should build the path to get here
        if node == END_NODE:
            return get_shortest_path_to_ship(previous_node_in_path)


        for connected_node in satellites[node]:
            # Check if we have visited this node yet
            if connected_node not in previous_node_in_path:

                # Add the node to the queue for our BFS
                bfs_node_queue.enqueue(connected_node)

                # Set the current node as the one that came before each of its siblings
                previous_node_in_path[connected_node] = node


    # We iterated through all the nodes, and no paths lead to the ship
    return None


def get_shortest_path_to_ship(previous_node_in_path):
    path_to_ship = []

    # We start at the final node and work our way backward to the start
    node = END_NODE

    while node != None:
        path_to_ship.append(node)

        # Doing a look up on the hash table for each node
        # will yield the previous node in the path
        node = previous_node_in_path[node]


    # Since we worked our way backward,
    # we must reverse the path to be in the correct order base -> ship
    return path_to_ship[::-1]

class Queue:
    def __init__(self):
        self.queue = deque()

    # deque appends and removes at opposite ends without shifting elements
    def enqueue(self, item):
        self.queue.append(item)

    def dequeue(self):
        return self.queue.popleft()

    def size(self):
        return len(self.queue)
