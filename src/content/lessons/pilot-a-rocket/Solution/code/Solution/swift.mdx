class Solution {
    private let startNode = "base"
    private let endNode = "ship"

    func pilotRocket(_ satellites: [String: [String]]) -> [String]? {
        // Queue to handle BFS
        var bfsNodeQueue = Queue<String>()
        bfsNodeQueue.enqueue(startNode)

        // Use a hash table to track the node that came
        // prior to the current in our BFS to track the path.
        // The start node has no previous node, so it's the one node never added.
        var previousNodeInPath: [String: String] = [:]

        // Perform our BFS
        while let node = bfsNodeQueue.dequeue() {
            // We've successfully reached the end, and now we should build the path to get here
            if node == endNode {
                return getShortestPathToShip(previousNodeInPath)
            }

            for connectedNode in satellites[node] ?? [] {
                // Check if we have visited this node yet
                if connectedNode != startNode && previousNodeInPath[connectedNode] == nil {

                    // Add the node to the queue for our BFS
                    bfsNodeQueue.enqueue(connectedNode)

                    // Set the current node as the one that came before each of its siblings
                    previousNodeInPath[connectedNode] = node
                }
            }
        }

        // We iterated through all the nodes, and no paths lead to the ship
        return nil
    }

    private func getShortestPathToShip(_ previousNodeInPath: [String: String]) -> [String] {
        var pathToShip: [String] = []

        // We start at the final node and work our way backward to the start
        var node: String? = endNode

        while let currentNode = node {
            pathToShip.append(currentNode)

            // Doing a look up on the hash table for each node
            // will yield the previous node in the path (nil for the start node)
            node = previousNodeInPath[currentNode]
        }

        // Since we worked our way backward,
        // we must reverse the path to be in the correct order base -> ship
        return pathToShip.reversed()
    }
}

// Swift's arrays remove from the front in O(n) time, so this queue keeps the
// index of its front item instead, which makes enqueue and dequeue both O(1)
struct Queue<Element> {
    private var items: [Element] = []
    private var front = 0

    var count: Int { return items.count - front }

    mutating func enqueue(_ item: Element) {
        items.append(item)
    }

    mutating func dequeue() -> Element? {
        guard front < items.count else { return nil }
        front += 1
        return items[front - 1]
    }
}
