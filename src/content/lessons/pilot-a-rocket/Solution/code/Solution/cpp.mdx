const string START_NODE = "base";
const string END_NODE = "ship";
// Marks the start node, which has no node before it in the path
const string NO_NODE = "";

class Solution {
public:
    optional<vector<string>> pilotRocket(unordered_map<string, vector<string>>& satellites) {
        // Queue to handle BFS
        queue<string> bfsNodeQueue;
        bfsNodeQueue.push(START_NODE);

        // Use a hash table to track the node that came
        // prior to the current in our BFS to track the path
        unordered_map<string, string> previousNodeInPath;
        previousNodeInPath[START_NODE] = NO_NODE;

        // Perform our BFS
        while (bfsNodeQueue.size() > 0) {
            string node = bfsNodeQueue.front();
            bfsNodeQueue.pop();

            // We've successfully reached the end, and now we should build the path to get here
            if (node == END_NODE) {
                return getShortestPathToShip(previousNodeInPath);
            }

            for (const string& connectedNode : satellites[node]) {
                // Check if we have visited this node yet
                if (!previousNodeInPath.contains(connectedNode)) {
                    // Add the node to the queue for our BFS
                    bfsNodeQueue.push(connectedNode);

                    // Set the current node as the one that came before each of its siblings
                    previousNodeInPath[connectedNode] = node;
                }
            }
        }

        // We iterated through all the nodes, and no paths lead to the ship
        return nullopt;
    }

private:
    vector<string> getShortestPathToShip(unordered_map<string, string>& previousNodeInPath) {
        vector<string> pathToShip;

        // We start at the final node and work our way backward to the start
        string node = END_NODE;

        while (node != NO_NODE) {
            pathToShip.push_back(node);

            // Doing a look up on the hash table for each node
            // will yield the previous node in the path
            node = previousNodeInPath[node];
        }

        // Since we worked our way backward,
        // we must reverse the path to be in the correct order base -> ship
        reverse(pathToShip.begin(), pathToShip.end());
        return pathToShip;
    }
};
