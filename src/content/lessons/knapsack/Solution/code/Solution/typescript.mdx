```javascript
interface Item {
  weight: number;
  value: number;
}

const getMaxValueKnapsack = (items: Item[], capacity: number): number => {
  /*
    maxValueAtWeight[i][weight] is the max value we can fit in a bag that holds `weight`
    using only the first i items. This is our dp / memo table for the subproblems.
    Row 0 means we haven't tried any items yet, so every weight starts at 0.
  */
  const maxValueAtWeight: number[][] = Array.from({ length: items.length + 1 }, () => new Array(capacity + 1).fill(0));

  // Try each item at every weight from 0 up to the capacity
  for (let i = 1; i <= items.length; i++) {
    const item = items[i - 1];

    for (let weight = 0; weight <= capacity; weight++) {
      // Option 1: skip the current item and keep the best value from the previous items
      const valueWithoutItem = maxValueAtWeight[i - 1][weight];
      maxValueAtWeight[i][weight] = valueWithoutItem;

      /*
        Option 2: if the item fits, take it and add its value to the best value
        the previous items can fit in the weight that is left over.
        We look at the previous row, so each item is only used once.
      */
      if (item.weight <= weight) {
        const valueWithItem = item.value + maxValueAtWeight[i - 1][weight - item.weight];
        maxValueAtWeight[i][weight] = Math.max(valueWithoutItem, valueWithItem);
      }
    }
  }

  // The last cell has tried every item at the full capacity
  return maxValueAtWeight[items.length][capacity];
};
```
