def is_bst(root):
    stack = []
    current_node = root
    in_order_value = float("-inf")

    # Begin itertaive in-order traversal
    while (len(stack) > 0) or (current_node != None):
        # Traverse down a node's left children as long as they continue to exist
        while current_node != None:
            stack.append(current_node)
            current_node = current_node.left

        # Check nodes in-order, testing that their value is larger than the previous
        current_node = stack.pop()

        # In-order traversal failed
        if current_node.data <= in_order_value:
            return False

        # Update value
        in_order_value = current_node.data

        # Continue in-order traveresal to the current node's right path
        current_node = current_node.right

    # Finish traversal and all values were valid
    return True
