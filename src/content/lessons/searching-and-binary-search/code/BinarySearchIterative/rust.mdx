// Iterative Binary Search
fn iterative_binary_search(array: &[i32], target: i32) -> i32 {
    // The bounds are signed, as the upper bound can move to before the first item
    let mut lower_bound = 0;
    let mut upper_bound = array.len() as i32 - 1;

    while lower_bound <= upper_bound {
        let midpoint_index = (lower_bound + upper_bound) / 2;
        let midpoint_item = array[midpoint_index as usize];

        if target == midpoint_item {
            return midpoint_index;
        }

        if midpoint_item < target {
            lower_bound = midpoint_index + 1;
        } else if midpoint_item > target {
            upper_bound = midpoint_index - 1;
        }
    }

    -1
}
