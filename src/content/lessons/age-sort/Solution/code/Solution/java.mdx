class Solution {
    private static final int MIN_AGE = 0;
    private static final int MAX_AGE = 100;

    public int[] sortByAge(int[] ageList) {
        int[] sortedAgeList = new int[ageList.length];
        // Java fills a new int array with 0s
        int[] ageCount = new int[MAX_AGE + 1];

        for (int age : ageList) {
            ageCount[age]++;
        }

        // The next spot to fill in the sorted list
        int sortedIndex = 0;

        for (int age = MIN_AGE; age <= MAX_AGE; age++) {
            int count = ageCount[age];

            for (int i = 0; i < count; i++) {
                sortedAgeList[sortedIndex] = age;
                sortedIndex++;
            }
        }

        return sortedAgeList;
    }
}
