class Solution {
    func findTargetIndex(_ items: [Int], _ target: Int) -> Int {
        // Initialize the upper and lower bound to the binary search
        var startIndex = 0
        var endIndex = items.count - 1

        // Loop until our bounds converge
        while startIndex <= endIndex {
            // Number of items between the lower and upper bound
            // (integer division rounds down here)
            let itemsBetweenStartEnd = (endIndex - startIndex) / 2
            let midIndex = startIndex + itemsBetweenStartEnd

            // Item found, return mid index as solution
            if items[midIndex] == target {
                return midIndex
            }

            let isLeftCorrectlyOrdered = items[midIndex] >= items[startIndex]

            // The left half, start to midpoint, is sorted normally
            // (rotation in right half or has been eliminated)
            if isLeftCorrectlyOrdered {
                let targetInLeftHalf = target >= items[startIndex] && target < items[midIndex]

                if targetInLeftHalf {
                    endIndex = midIndex - 1
                } else {
                    startIndex = midIndex + 1
                }
            // The right half, midpoint to the end, is sorted normally
            // (rotation in left half)
            } else {
                let targetInRightHalf = target <= items[endIndex] && target > items[midIndex]

                if targetInRightHalf {
                    startIndex = midIndex + 1
                } else {
                    endIndex = midIndex - 1
                }
            }
        }

        // Target not found in the input
        return -1
    }
}
