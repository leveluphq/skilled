```swift
var memo: [Int: Int] = [:]

func fib(_ n: Int) -> Int {
    if n < 2 {
        return n
    }

    // If we have the value in our hash table,
    // return it instead of continuing with the recursive case
    if let memoized = memo[n] {
        return memoized
    }

    let nthFib = fib(n - 1) + fib(n - 2)

    memo[n] = nthFib

    return nthFib
}
```
