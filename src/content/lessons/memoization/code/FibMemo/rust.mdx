```rust
// A Rust function can't use a variable from outside it, so we pass the memo in
let mut memo: HashMap<u64, u64> = HashMap::new();

fn fib(n: u64, memo: &mut HashMap<u64, u64>) -> u64 {
    if n < 2 {
        return n;
    }

    // If we have the value in our hash table,
    // return it instead of continuing with the recursive case
    if let Some(&nth_fib) = memo.get(&n) {
        return nth_fib;
    }

    let nth_fib = fib(n - 1, memo) + fib(n - 2, memo);

    memo.insert(n, nth_fib);

    nth_fib
}
```
