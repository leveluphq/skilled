```javascript
const memo = {};

const fib = (n: number): number => {
  if (n < 2) {
    return n;
  }

  // If we have the value in our hash table,
  // return it instead of continuing with the recursive case
  if (memo[n]) {
    return memo[n];
  }

  const nthFib = fib(n - 1) + fib(n - 2);

  memo[n] = nthFib;

  return nthFib;
}
```
