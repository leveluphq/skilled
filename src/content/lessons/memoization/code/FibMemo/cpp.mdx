unordered_map<int, int> memo;

int fib(int n) {
    if (n < 2) {
        return n;
    }

    // If we have the value in our hash table,
    // return it instead of continuing with the recursive case
    if (memo.contains(n)) {
        return memo[n];
    }

    int nthFib = fib(n - 1) + fib(n - 2);

    memo[n] = nthFib;

    return nthFib;
}
