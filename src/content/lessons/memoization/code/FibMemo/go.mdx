```go
var memo = map[int]int{}

func fib(n int) int {
	if n < 2 {
		return n
	}

	// If we have the value in our hash table,
	// return it instead of continuing with the recursive case
	if value, ok := memo[n]; ok {
		return value
	}

	nthFib := fib(n-1) + fib(n-2)

	memo[n] = nthFib

	return nthFib
}
```
