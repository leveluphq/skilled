Dictionary<int, long> memo = new();

long Fib(int n) {
    if (n < 2) {
        return n;
    }

    // If we have the value in our hash table,
    // return it instead of continuing with the recursive case
    if (memo.ContainsKey(n)) {
        return memo[n];
    }

    long nthFib = Fib(n - 1) + Fib(n - 2);

    memo[n] = nthFib;

    return nthFib;
}
