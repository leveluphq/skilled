public TValue Get(string key) {
    int index = Hash(key);
    List<Entry> bucket = table[index];

    // Check if the key exists in the bucket, and return its value if found
    if (bucket != null) {
        for (int i = 0; i < bucket.Count; i++) {
            if (bucket[i].key == key) {
                return bucket[i].value;
            }
        }
    }

    // Return the default value (null for reference types) if the key is not found
    return default;
}
