```go
import "slices"

// A key-value pair stored in a bucket
type entry struct {
	key   string
	value any
}

type HashTable struct {
	size  int
	table [][]entry
}

func NewHashTable(size int) *HashTable {
	return &HashTable{
		size:  size,
		table: make([][]entry, size),
	}
}

func (ht *HashTable) Get(key string) any {
	index := ht.Hash(key)
	bucket := ht.table[index]

	for i := range bucket {
		if bucket[i].key == key {
			return bucket[i].value
		}
	}

	return nil
}

func (ht *HashTable) Insert(key string, value any) {
	index := ht.Hash(key)

	if ht.table[index] == nil {
		ht.table[index] = []entry{}
	}

	bucket := ht.table[index]

	for i := range bucket {
		if bucket[i].key == key {
			bucket[i].value = value
			return
		}
	}

	ht.table[index] = append(bucket, entry{key, value})
}

func (ht *HashTable) Remove(key string) any {
	index := ht.Hash(key)
	bucket := ht.table[index]

	for i := range bucket {
		if bucket[i].key == key {
			value := bucket[i].value
			ht.table[index] = slices.Delete(bucket, i, i+1)

			return value
		}
	}

	return nil
}

func (ht *HashTable) Hash(key string) int {
	hash := 0

	for _, char := range key {
		hash = (hash + int(char)) % ht.size
	}

	return hash
}
```
