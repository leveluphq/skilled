```swift
import Foundation

class HashTable {
    let size: Int
    var table: [[(key: String, value: Int)]]

    init(_ size: Int) {
        self.size = size
        self.table = Array(repeating: [], count: size)
    }

    func get(_ key: String) -> Int? {
        let index = hash(key)
        let bucket = table[index]

        for i in 0..<bucket.count {
            if bucket[i].key == key {
                return bucket[i].value
            }
        }

        return nil
    }

    func insert(_ key: String, _ value: Int) {
        let index = hash(key)

        for i in 0..<table[index].count {
            if table[index][i].key == key {
                table[index][i].value = value
                return
            }
        }

        table[index].append((key: key, value: value))
    }

    func remove(_ key: String) -> Int? {
        let index = hash(key)

        for i in 0..<table[index].count {
            if table[index][i].key == key {
                let value = table[index][i].value
                table[index].remove(at: i)

                return value
            }
        }

        return nil
    }

    func hash(_ key: String) -> Int {
        var hash = 0

        // Equal Swift strings must hash to the same bucket.
        for character in key.precomposedStringWithCanonicalMapping.unicodeScalars {
            hash = (hash + Int(character.value)) % size
        }

        return hash
    }
}
```
