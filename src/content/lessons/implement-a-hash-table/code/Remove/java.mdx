public V remove(String key) {
    int index = hash(key);
    List<Entry> bucket = table[index];

    // Find the key in the bucket, delete the key-value pair, and return its value
    if (bucket != null) {
        for (int i = 0; i < bucket.size(); i++) {
            if (bucket.get(i).key.equals(key)) {
                V value = bucket.get(i).value;
                bucket.remove(i);

                return value;
            }
        }
    }

    // Return null if the key didn't exist
    return null;
}
