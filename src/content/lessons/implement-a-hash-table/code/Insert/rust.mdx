fn insert(&mut self, key: &str, value: i32) {
    let index = self.hash(key);

    // Initialize our bucket if it doesn't exist yet
    let bucket = self.table[index].get_or_insert_with(Vec::new);

    // If the key already exists in the bucket, replace its value
    for (bucket_key, bucket_value) in bucket.iter_mut() {
        if bucket_key == key {
            *bucket_value = value;
            return;
        }
    }

    // Otherwise, add the new key-value pair
    bucket.push((key.to_string(), value));
}
