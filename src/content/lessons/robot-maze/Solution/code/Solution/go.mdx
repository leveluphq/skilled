const (
	open    = 0
	blocked = 1
	exit    = 2
)

// What the memo knows about the paths from a node
const (
	notChecked = iota
	reachesExit
	deadEnd
)

func hasExitPath(roomMatrix [][]int) bool {
	// Return false if the matrix is invalid
	if len(roomMatrix) == 0 || len(roomMatrix[0]) == 0 {
		return false
	}

	// Init our memoize matrix, where no node has been checked yet
	memoizedPathValidity := make([][]int, len(roomMatrix))
	for row := range memoizedPathValidity {
		memoizedPathValidity[row] = make([]int, len(roomMatrix[0]))
	}

	// Recursively walk through the room to check if there is a valid path
	isValidPath := walkThroughRoom(roomMatrix, 0, 0, memoizedPathValidity)

	return isValidPath
}

// Recursive helper function
func walkThroughRoom(roomMatrix [][]int, row int, col int, memo [][]int) bool {
	// Base case 1: Outside of room, so there can't be a path through this
	isOutsideRoom := row > len(roomMatrix)-1 || col > len(roomMatrix[0])-1
	if isOutsideRoom {
		return false
	}

	// Base case 2: A path successfully reached the end, so propagate this upward
	isOnValidPath := memo[row][col] == reachesExit
	isExitNode := roomMatrix[row][col] == exit
	if isOnValidPath || isExitNode {
		// Set the current node as reaching the exit as well
		memo[row][col] = reachesExit
		return true
	}

	// Base case 3: We reached a blocking node
	isOnInvalidPath := memo[row][col] == deadEnd
	isBlockingNode := roomMatrix[row][col] == blocked
	if isOnInvalidPath || isBlockingNode {
		memo[row][col] = deadEnd
		return false
	}

	// Recursive case: move down and right
	downResult := walkThroughRoom(roomMatrix, row+1, col, memo)
	rightResult := walkThroughRoom(roomMatrix, row, col+1, memo)

	if !downResult && !rightResult {
		// If both this node's right and down are false, we memoize this as a dead end
		memo[row][col] = deadEnd
	} else {
		// If either of the paths are true, this node reaches the exit too
		memo[row][col] = reachesExit
	}

	// Returns true if either is successful
	return downResult || rightResult
}
