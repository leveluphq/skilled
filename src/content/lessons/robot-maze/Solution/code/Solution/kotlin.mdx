const val OPEN = 0
const val BLOCKED = 1
const val EXIT = 2

class Solution {
    fun hasExitPath(roomMatrix: Array<IntArray>): Boolean {
        // Return false if the matrix is invalid
        if (roomMatrix.isEmpty() || roomMatrix[0].isEmpty()) {
            return false
        }

        // Init our memoize matrix: null until we know whether a cell is on a valid path
        val memoizedPathValidity = Array(roomMatrix.size) { arrayOfNulls<Boolean>(roomMatrix[0].size) }

        // Recursively walk through the room to check if there is a valid path
        val isValidPath = walkThroughRoom(roomMatrix, 0, 0, memoizedPathValidity)

        return isValidPath
    }

    // Recursive helper function
    private fun walkThroughRoom(roomMatrix: Array<IntArray>, row: Int, col: Int, memo: Array<Array<Boolean?>>): Boolean {
        // Base case 1: Outside of room, so there can't be a path through this
        val isOutsideRoom = row > roomMatrix.size - 1 || col > roomMatrix[0].size - 1
        if (isOutsideRoom) {
            return false
        }

        // Base case 2: A path successfully reached the end, so propagate this upward
        val isOnValidPath = memo[row][col] == true
        val isExitNode = roomMatrix[row][col] == EXIT
        if (isOnValidPath || isExitNode) {
            // Set the current node as true as well
            memo[row][col] = true
            return true
        }

        // Base case 3: We reached a blocking node
        val isOnInvalidPath = memo[row][col] == false
        val isBlockingNode = roomMatrix[row][col] == BLOCKED
        if (isOnInvalidPath || isBlockingNode) {
            memo[row][col] = false
            return false
        }

        // Recursive case: move down and right
        val downResult = walkThroughRoom(roomMatrix, row + 1, col, memo)
        val rightResult = walkThroughRoom(roomMatrix, row, col + 1, memo)

        if (!downResult && !rightResult) {
            // If both this node's right and down are false, we memoize this to false
            memo[row][col] = false
        } else {
            // If either of the paths are true, we set this node as true since at least one can reach the exit
            memo[row][col] = true
        }

        // Returns true if either is successful
        return downResult || rightResult
    }
}
