// Open cells are 0
const BLOCKED: i32 = 1;
const EXIT: i32 = 2;

impl Solution {
    pub fn has_exit_path(room_matrix: Vec<Vec<i32>>) -> bool {
        // Return false if the matrix is invalid
        if room_matrix.is_empty() || room_matrix[0].is_empty() {
            return false;
        }

        // Init our memoize matrix: None until we know if a cell is on a path to the exit
        let mut memoized_path_validity = vec![vec![None; room_matrix[0].len()]; room_matrix.len()];

        // Recursively walk through the room to check if there is a valid path
        walk_through_room(&room_matrix, 0, 0, &mut memoized_path_validity)
    }
}

// Recursive helper function
fn walk_through_room(
    room_matrix: &[Vec<i32>],
    row: usize,
    col: usize,
    memo: &mut [Vec<Option<bool>>],
) -> bool {
    // Base case 1: Outside of room, so there can't be a path through this
    let is_outside_room = row >= room_matrix.len() || col >= room_matrix[0].len();
    if is_outside_room {
        return false;
    }

    // Base case 2: A path successfully reached the end, so propagate this upward
    let is_on_valid_path = memo[row][col] == Some(true);
    let is_exit_node = room_matrix[row][col] == EXIT;
    if is_on_valid_path || is_exit_node {
        // Set the current node as true as well
        memo[row][col] = Some(true);
        return true;
    }

    // Base case 3: We reached a blocking node
    let is_on_invalid_path = memo[row][col] == Some(false);
    let is_blocking_node = room_matrix[row][col] == BLOCKED;
    if is_on_invalid_path || is_blocking_node {
        memo[row][col] = Some(false);
        return false;
    }

    // Recursive case: move down and right
    let down_result = walk_through_room(room_matrix, row + 1, col, memo);
    let right_result = walk_through_room(room_matrix, row, col + 1, memo);

    // If both this node's right and down are false, we memoize this to false.
    // If either of the paths are true, we set this node as true since at least one can reach the exit
    memo[row][col] = Some(down_result || right_result);

    // Returns true if either is successful
    down_result || right_result
}
