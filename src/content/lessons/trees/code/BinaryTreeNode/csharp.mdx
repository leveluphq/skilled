public class BinaryTreeNode {
    // NOTE: We're defining the data as an int here, but in reality it can be any data type
    public int data;
    // left and right point to either null or another BinaryTreeNode
    public BinaryTreeNode left;
    public BinaryTreeNode right;

    public BinaryTreeNode(int data) {
        this.data = data;
        this.left = null;
        this.right = null;
    }
}
