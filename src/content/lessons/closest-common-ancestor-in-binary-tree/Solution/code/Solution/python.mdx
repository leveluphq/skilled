def find_closest_ancestor(root, first_child_value, second_child_value):
    # Base case 1: We recursed to the end of a path
    if root == None:
        return None

    # Base case 2: We found one of the target child nodes
    if (root.data == first_child_value) or (root.data == second_child_value):
        return root

    """
        Continue searching left and right.
        This ensures we check all the nodes below this one recursively before we handle the current value.
        This is a post-order traversal.
    """
    left_search_result = find_closest_ancestor(root.left, first_child_value, second_child_value)
    right_search_result = find_closest_ancestor(root.right, first_child_value, second_child_value)

    """
        We found the common ancestor node.
        Return it, and it will propagate up and be the solution.
    """
    if (left_search_result != None) and (right_search_result != None):
        return root

    """
        Return whichever node isn't null.
        If they both are null, it doesn't matter which we return.
    """
    return left_search_result if left_search_result != None else right_search_result
