class Solution {
    func findClosestAncestor(_ root: BinaryNode?, _ firstChildValue: Int, _ secondChildValue: Int) -> BinaryNode? {
        // Base case 1: We recursed to the end of a path
        guard let root else {
            return nil
        }

        // Base case 2: We found one of the target child nodes
        if root.data == firstChildValue || root.data == secondChildValue {
            return root
        }

        /*
          Continue searching left and right.
          This ensures we check all the nodes below this one recursively before we handle the current value.
          This is a post-order traversal.
        */
        let leftSearchResult = findClosestAncestor(root.left, firstChildValue, secondChildValue)
        let rightSearchResult = findClosestAncestor(root.right, firstChildValue, secondChildValue)

        /*
          We found the common ancestor node.
          Return it, and it will propagate up and be the solution.
        */
        if leftSearchResult != nil && rightSearchResult != nil {
            return root
        }

        /*
          Return whichever node isn't nil.
          If they both are nil, it doesn't matter which we return.
        */
        return leftSearchResult ?? rightSearchResult
    }
}
