func findClosestAncestor(root *BinaryNode, firstChildValue int, secondChildValue int) *BinaryNode {
	// Base case 1: We recursed to the end of a path
	if root == nil {
		return nil
	}

	// Base case 2: We found one of the target child nodes
	if root.Data == firstChildValue || root.Data == secondChildValue {
		return root
	}

	// Continue searching left and right.
	// This ensures we check all the nodes below this one recursively before we
	// handle the current value. This is a post-order traversal.
	leftSearchResult := findClosestAncestor(root.Left, firstChildValue, secondChildValue)
	rightSearchResult := findClosestAncestor(root.Right, firstChildValue, secondChildValue)

	// We found the common ancestor node.
	// Return it, and it will propagate up and be the solution.
	if leftSearchResult != nil && rightSearchResult != nil {
		return root
	}

	// Return whichever node isn't nil.
	// If they both are nil, it doesn't matter which we return.
	if leftSearchResult != nil {
		return leftSearchResult
	}
	return rightSearchResult
}
