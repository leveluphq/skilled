class Solution {
public:
    vector<vector<int>> getTargetIndexes(vector<int>& array, int target) {
        vector<vector<int>> results;
        unordered_map<int, int> arrayElementAtIndex;

        // Loop through our input array and
        // create a hash table that stores the array value and
        // its corresponding index.
        for (int i = 0; i < (int) array.size(); i++) {
            int element = array[i];
            arrayElementAtIndex[element] = i;
        }

        for (int i = 0; i < (int) array.size(); i++) {
            // We find the element that would be the match current value
            // and check if it is our hash table.
            int currentValue = array[i];
            int matchedValue = target - currentValue;
            auto match = arrayElementAtIndex.find(matchedValue);

            // Check if our matched index is a valid array index
            // ----
            // NOTE: find() returns end() when the value isn't
            // in the hash table, which we treat as index -1.
            int matchedIndex = match == arrayElementAtIndex.end() ? -1 : match->second;

            if (matchedIndex >= 0 && matchedIndex != i) {
                results.push_back({i, matchedIndex});

                // Make sure we don't use these values again
                arrayElementAtIndex[currentValue] = -1;
                arrayElementAtIndex[matchedValue] = -1;
            }
        }

        return results;
    }
};
