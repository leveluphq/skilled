public int[][] bruteForceGetTargetIndexes(int[] array, int target) {
    List<int[]> results = new ArrayList<>();

    // Iterate through array to check each value against every other value
    for (int i = 0; i < array.length; i++) {
        // We start j at i + 1 to make sure we don't compare the same index pairs more than once
        for (int j = i + 1; j < array.length; j++) {
            // If the sum of the values equals the target, we add the pair to the results as an array
            if (array[i] + array[j] == target) {
                results.add(new int[] { i, j });
            }
        }
    }

    return results.toArray(new int[0][]);
}
