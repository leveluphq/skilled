// #1
function printHello(input) {
  let ten = 5 + 5;
  let twenty = 2 * ten;

  for (let i = 0; i < twenty; i++) {
    console.log('Hello, world!');
  }
}

// #2
function printInput(input) {
  const n = input.length;
  for (let i = 0; i < n; i++) {
    console.log(i);
  }
}

// #3
function printNestedInput(input) {
  const n = input.length;
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n; j++) {
      console.log(i * j);
    }
  }
}
