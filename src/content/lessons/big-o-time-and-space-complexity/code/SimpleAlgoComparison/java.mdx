// #1
static void printHello(int[] input) {
    int ten = 5 + 5;
    int twenty = 2 * ten;

    for (int i = 0; i < twenty; i++) {
        System.out.println("Hello, world!");
    }
}

// #2
static void printInput(int[] input) {
    int n = input.length;
    for (int i = 0; i < n; i++) {
        System.out.println(i);
    }
}

// #3
static void printNestedInput(int[] input) {
    int n = input.length;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            System.out.println(i * j);
        }
    }
}
