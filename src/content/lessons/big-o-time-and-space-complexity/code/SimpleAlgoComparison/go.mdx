// #2
func printInput(input []int) {
	n := len(input)
	for i := 0; i < n; i++ {
		fmt.Println(i)
	}
}

// #3
func printNestedInput(input []int) {
	n := len(input)
	for i := 0; i < n; i++ {
		for j := 0; j < n; j++ {
			fmt.Println(i * j)
		}
	}
}
