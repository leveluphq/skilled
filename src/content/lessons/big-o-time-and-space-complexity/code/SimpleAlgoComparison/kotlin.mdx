// #1
fun printHello(input: IntArray) {
    val ten = 5 + 5
    val twenty = 2 * ten

    for (i in 0 until twenty) {
        println("Hello, world!")
    }
}

// #2
fun printInput(input: IntArray) {
    val n = input.size
    for (i in 0 until n) {
        println(i)
    }
}

// #3
fun printNestedInput(input: IntArray) {
    val n = input.size
    for (i in 0 until n) {
        for (j in 0 until n) {
            println(i * j)
        }
    }
}
