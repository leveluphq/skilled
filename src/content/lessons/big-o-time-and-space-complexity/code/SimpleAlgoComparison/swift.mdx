```swift
// #1
func printHello(_ input: [Int]) {
    let ten = 5 + 5
    let twenty = 2 * ten

    for _ in 0..<twenty {
        print("Hello, world!")
    }
}

// #2
func printInput(_ input: [Int]) {
    let n = input.count
    for i in 0..<n {
        print(i)
    }
}

// #3
func printNestedInput(_ input: [Int]) {
    let n = input.count
    for i in 0..<n {
        for j in 0..<n {
            print(i * j)
        }
    }
}
```
