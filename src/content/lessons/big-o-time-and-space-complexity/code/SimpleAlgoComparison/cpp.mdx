// #2
void printInput(const vector<int>& input) {
    int n = input.size();
    for (int i = 0; i < n; i++) {
        cout << i << endl;
    }
}

// #3
void printNestedInput(const vector<int>& input) {
    int n = input.size();
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << i * j << endl;
        }
    }
}
