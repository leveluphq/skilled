```python
n = 1000000000 # or some large number
array_of_n = [None for _ in range(n)]

# O(n^3)
for i in range(n):
    for j in range(n):
        for k in range(n):
            # ...

# O(n^3)
for i in array_of_n:
    for j in array_of_n:
        for k in array_of_n:
            # ...

# O(n)
for i in range(n):
    # ...

for j in range(n):
    # ...

for k in range(n):
    # ...

# Nested is 1000000000 ^ 3 = 1000000000000000000000000000
# Sequential is 3000000000
# Sequential is MUCH MUCH smaller
```
