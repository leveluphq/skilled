```javascript
const n = 1000000000; // or some large number
const arrayOfN = new Array(n);

// O(n^3)
for (let i = 0; i < n; i++) {
  for (let j = 0; j < n; j++) {
    for (let k = 0; k < n; k++) {
      // ...
    }
  }
}

// O(n^3)
arrayOfN.forEach(i => {
  arrayOfN.forEach(j => {
    arrayOfN.forEach(k => {
      // ...
    });
  });
});

// O(n)
for (let i = 0; i < n; i++) {
  // ...
}
for (let j = 0; j < n; j++) {
  // ...
}
for (let k = 0; k < n; k++) {
  // ...
}

// Nested is 1000000000 ^ 3 = 1000000000000000000000000000
// Sequential is 3000000000
// Sequential is MUCH MUCH smaller
```
