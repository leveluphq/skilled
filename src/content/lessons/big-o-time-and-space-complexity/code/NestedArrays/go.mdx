```go
n := 1000000000 // or some large number
arrayOfN := make([]int, n)

// O(n^3)
for i := 0; i < n; i++ {
	for j := 0; j < n; j++ {
		for k := 0; k < n; k++ {
			// ...
		}
	}
}

// O(n^3)
for range arrayOfN {
	for range arrayOfN {
		for range arrayOfN {
			// ...
		}
	}
}

// O(n)
for i := 0; i < n; i++ {
	// ...
}
for j := 0; j < n; j++ {
	// ...
}
for k := 0; k < n; k++ {
	// ...
}

// Nested is 1000000000 ^ 3 = 1000000000000000000000000000
// Sequential is 3000000000
// Sequential is MUCH MUCH smaller
```
