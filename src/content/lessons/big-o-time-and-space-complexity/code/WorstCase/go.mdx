func findFour(input []int) int {
	for i := 0; i < len(input); i++ {
		if input[i] == 4 {
			return i
		}
	}
	return -1
}

// Average Case: find 6 in the middle at index 3 - iterate 4 items
func findSix(input []int) int {
	for i := 0; i < len(input); i++ {
		if input[i] == 6 {
			return i
		}
	}
	return -1
}

// Worst Case: find 3 at the last index - iterate all `n` items
func findThree(input []int) int {
	for i := 0; i < len(input); i++ {
		if input[i] == 3 {
			return i
		}
	}
	return -1
}
