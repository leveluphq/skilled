arr = [4, 5, 7, 6, 2, 0, 1, 3]

def find_four(input):
    for i in range(len(input)):
        if input[i] == 4:
            return i

# Average Case: find 6 in the middle at index 3 - iterate 4 items
def find_six(input):
    for i in range(len(input)):
        if input[i] == 6:
            return i

# Worst Case: find 3 at the last index - iterate all `n` items
def find_three(input):
    for i in range(len(input)):
        if input[i] == 3:
            return i
