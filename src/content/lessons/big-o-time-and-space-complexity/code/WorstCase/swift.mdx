        if input[i] == 4 {
            return i
        }
    }

}

func findSix(_ input: [Int]) -> Int? {
    for i in 0..<input.count {
        if input[i] == 6 {
            return i
        }
    }

    return nil
}

// Worst Case: find 3 at the last index - iterate all `n` items
func findThree(_ input: [Int]) -> Int? {
    for i in 0..<input.count {

        }
