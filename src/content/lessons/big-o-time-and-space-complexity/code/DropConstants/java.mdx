static void case1(int[] input) {
    for (int i = 0; i < input.length; i++) {
        System.out.println(i * 2); // O(n)
    }
    for (int i = 0; i < input.length; i++) {
        System.out.println(i * 10); // O(n)
    }
    for (int i = 0; i < input.length; i++) {
        System.out.println(i * 42); // O(n)
    }
}

static void case2(int[] input) {

        System.out.println(i * 2); // O(n)

        System.out.println(i * 42); // O(n)
    }

static void case3(int[] input) {
    for (int i = 0; i < input.length; i++) {
        for (int j = 0; j < input.length; j++) {
            System.out.println(i + " " + j);
        }
    }
}
