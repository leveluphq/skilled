```cpp
// sort() sorts a range in place, and it compares items with < by default,
// so numbers sort by their value
vector<int> numbers = {1, 50, 10000, 510, 5050, 60};
sort(numbers.begin(), numbers.end()); // {1, 50, 60, 510, 5050, 10000}

// Pass a comparison function to change the order
// It returns true when a must come before b (a bool, not a negative/zero/positive number)
sort(numbers.begin(), numbers.end(), greater<int>()); // {10000, 5050, 510, 60, 50, 1}

// Strings sort as expected
vector<string> months = {"March", "Jan", "Feb", "Dec"};
sort(months.begin(), months.end()); // {"Dec", "Feb", "Jan", "March"}

// Use a lambda function to sort structs by a field
struct Player {
    int score;
};
vector<Player> players = {{20}, {100}, {5}};
sort(players.begin(), players.end(), [](const Player& a, const Player& b) {
    return a.score < b.score;
});
// => {{5}, {20}, {100}}
```
