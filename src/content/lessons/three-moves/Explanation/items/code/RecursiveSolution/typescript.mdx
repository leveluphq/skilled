function countTotalMoves(boardLength: number): number {
  // Base case 1: Invalid board length
  if (boardLength < 0) {
    return 0;
  // Base case 2: Introduction says length 0 is still 1 step
  } else if (boardLength === 0) {
    return 1;
  } else {
    // Recursive case
    return countTotalMoves(boardLength - 1) +
      countTotalMoves(boardLength - 2) +
      countTotalMoves(boardLength - 3);
  }
};
