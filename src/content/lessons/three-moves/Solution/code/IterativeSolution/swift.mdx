class Solution {
    func countTotalMoves(_ boardLength: Int) -> Int {
        if boardLength < 0 {
            return 0
        }

        var threeSpotsBack = 0
        var twoSpotsBack = 0
        var oneSpotBack = 1

        for _ in 0..<boardLength {
            let finalSpot = oneSpotBack + twoSpotsBack + threeSpotsBack

            threeSpotsBack = twoSpotsBack
            twoSpotsBack = oneSpotBack
            oneSpotBack = finalSpot
        }

        // We return oneSpotBack because it was actually updated to the finalSpot
        // during the last iteration of the loop when we moved values forward
        return oneSpotBack
    }
}
