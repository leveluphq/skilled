class Solution {
public:
    int countTotalMoves(int boardLength) {
        if (boardLength < 0) return 0;

        int threeSpotsBack = 0;
        int twoSpotsBack = 0;
        int oneSpotBack = 1;

        for (int i = 0; i < boardLength; i++) {
            int finalSpot = oneSpotBack + twoSpotsBack + threeSpotsBack;

            threeSpotsBack = twoSpotsBack;
            twoSpotsBack = oneSpotBack;
            oneSpotBack = finalSpot;
        }

        // We return oneSpotBack because it was actually updated to the finalSpot
        // during the last iteration of the loop when we moved values forward
        return oneSpotBack;
    }
};
