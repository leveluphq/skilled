public class Solution {
    // We will use -1 to indicate an unfilled spot
    private const int UncountedSpot = -1;

    public int CountTotalMoves(int boardLength) {
        if (boardLength < 0) {
            return 0;
        }

        // Initialize our memoize array
        var memo = new int[boardLength + 1];
        Array.Fill(memo, UncountedSpot);
        int totalMoves = CountTotalMovesMemoized(boardLength, memo);
        return totalMoves;
    }

    // Recursive helper function that also tracks memoized values
    private int CountTotalMovesMemoized(int boardLength, int[] memo) {
        // Base case 1: Invalid board length
        if (boardLength < 0) {
            return 0;
        // Base case 2: Prompt says length 0 is still 1 step
        } else if (boardLength == 0) {
            return 1;
        } else if (memo[boardLength] != UncountedSpot) {
            return memo[boardLength];
        } else {
            // Recursive case (and memoize its result)
            memo[boardLength] = CountTotalMovesMemoized(boardLength - 1, memo) +
                CountTotalMovesMemoized(boardLength - 2, memo) +
                CountTotalMovesMemoized(boardLength - 3, memo);

            return memo[boardLength];
        }
    }
}
