class Solution {
    func isValidSudokuBoard(_ board: [[Int]]) -> Bool {
        // Exit early if the board does not have a valid shape
        if !hasValidShape(board) {
            return false
        }

        return hasValidCells(board)
    }

    // The function to check for a valid shape is all constant time and space O(1)
    private func hasValidShape(_ board: [[Int]]) -> Bool {
        // We have an empty board, it is automatically invalid
        if board.isEmpty || board[0].isEmpty {
            return false
        }

        let rowCount = board.count
        let columnCount = board[0].count
        let isSquare = rowCount == columnCount

        // The board is not a square, so we know it is invalid
        if !isSquare {
            return false
        }

        // For a board to be composed of sub-squares,
        // the square root of `k` must be a whole number
        let hasSubsquares = Double(rowCount).squareRoot().truncatingRemainder(dividingBy: 1) == 0
        if !hasSubsquares {
            return false
        }

        // All conditions are met, so it is a valid shape
        return true
    }

    private func hasValidCells(_ board: [[Int]]) -> Bool {
        // Get the dimensions of the board and sub-squares
        let sizeOfBoard = board.count
        let sizeOfSubsquare = Int(Double(sizeOfBoard).squareRoot())

        // Create our one-dimensional array of hash tables for rows and columns
        // (dictionaries are values in Swift, so each spot gets its own hash table)
        var rows = [[Int: Int]](repeating: [:], count: sizeOfBoard)
        var columns = [[Int: Int]](repeating: [:], count: sizeOfBoard)

        // Create a two-dimensional array of hash tables
        // where there is a separate hash table for each sub-square
        var subsquares = [[[Int: Int]]](
            repeating: [[Int: Int]](repeating: [:], count: sizeOfSubsquare),
            count: sizeOfSubsquare
        )

        for row in 0..<sizeOfBoard {
            for col in 0..<sizeOfBoard {
                let cellValue = board[row][col]

                // If we have a value, we need to add it to our hash tables
                // and return false if a value already existed.
                if cellValue > 0 {
                    // Since the sub-square tracker is a 2D jxj array,
                    // we need to normalize the coordinates by j
                    // (integer division rounds down here)
                    let subsquareRow = row / sizeOfSubsquare
                    let subsquareCol = col / sizeOfSubsquare

                    // Initialize and/or increment the cell value for the corresponding row, column, sub-square
                    //
                    // rows and columns are a 1-D array where we look for the cellValue in a hash table
                    rows[row][cellValue, default: 0] += 1
                    columns[col][cellValue, default: 0] += 1
                    //
                    // subsquares is a 2-D matrix
                    subsquares[subsquareRow][subsquareCol][cellValue, default: 0] += 1

                    // If the value is greater than 1, we have already seen this in the hash table
                    let hasRowDuplicate = rows[row][cellValue, default: 0] > 1
                    let hasColumnDuplicate = columns[col][cellValue, default: 0] > 1
                    let hasSubsquareDuplicate = subsquares[subsquareRow][subsquareCol][cellValue, default: 0] > 1

                    // Return if there are any duplicates
                    if hasRowDuplicate || hasColumnDuplicate || hasSubsquareDuplicate {
                        return false
                    }
                }
            }
        }

        // We iterated to the end and didn't find any duplicates
        return true
    }
}
