class Solution {
    public boolean isValidSudokuBoard(int[][] board) {
        // Exit early if the board does not have a valid shape
        if (!hasValidShape(board)) {
            return false;
        }

        return hasValidCells(board);
    }

    // The function to check for a valid shape is all constant time and space O(1)
    private boolean hasValidShape(int[][] board) {
        // We have an empty board, it is automatically invalid
        if (board.length == 0 || board[0].length == 0) {
            return false;
        }

        int rowCount = board.length;
        int columnCount = board[0].length;
        boolean isSquare = rowCount == columnCount;

        // The board is not a square, so we know it is invalid
        if (!isSquare) {
            return false;
        }

        // For a board to be composed of sub-squares,
        // the square root of `k` must be a whole number
        boolean hasSubsquares = Math.sqrt(rowCount) % 1 == 0;
        if (!hasSubsquares) {
            return false;
        }

        // All conditions are met, so it is a valid shape
        return true;
    }

    // Helper function to init our list of hash tables
    private List<Map<Integer, Integer>> listOfMaps(int size) {
        List<Map<Integer, Integer>> maps = new ArrayList<>();
        for (int i = 0; i < size; i++) {
            maps.add(new HashMap<>());
        }
        return maps;
    }

    private List<List<Map<Integer, Integer>>> nestedListOfMaps(int size) {
        List<List<Map<Integer, Integer>>> lists = new ArrayList<>();
        for (int i = 0; i < size; i++) {
            lists.add(listOfMaps(size));
        }
        return lists;
    }

    private boolean hasValidCells(int[][] board) {
        // Get the dimensions of the board and sub-squares
        int sizeOfBoard = board.length;
        int sizeOfSubsquare = (int) Math.sqrt(sizeOfBoard);

        // Create our one-dimensional list of hash tables for rows and columns
        List<Map<Integer, Integer>> rows = listOfMaps(sizeOfBoard);
        List<Map<Integer, Integer>> columns = listOfMaps(sizeOfBoard);

        // Create a two-dimensional list of hash tables
        // where there is a separate hash table for each sub-square
        List<List<Map<Integer, Integer>>> subsquares = nestedListOfMaps(sizeOfSubsquare);

        for (int row = 0; row < sizeOfBoard; row++) {
            for (int col = 0; col < sizeOfBoard; col++) {
                int cellValue = board[row][col];

                // If we have a value, we need to add it to our hash tables
                // and return false if a value already existed.
                if (cellValue > 0) {
                    // Since the sub-square tracker is a 2D jxj list,
                    // we need to normalize the coordinates by j
                    int subsquareRow = row / sizeOfSubsquare;
                    int subsquareCol = col / sizeOfSubsquare;

                    // Initialize and/or increment the cell value for the corresponding row, column, sub-square.
                    // merge starts the count at 1 or adds 1 to it, and returns the new count.
                    //
                    // rows and columns are a 1-D list where we look for the cellValue in a hash table
                    int rowCount = rows.get(row).merge(cellValue, 1, Integer::sum);
                    int columnCount = columns.get(col).merge(cellValue, 1, Integer::sum);
                    //
                    // subsquares is a 2-D matrix
                    int subsquareCount = subsquares.get(subsquareRow).get(subsquareCol).merge(cellValue, 1, Integer::sum);

                    // If the value is greater than 1, we have already seen this in the hash table
                    boolean hasRowDuplicate = rowCount > 1;
                    boolean hasColumnDuplicate = columnCount > 1;
                    boolean hasSubsquareDuplicate = subsquareCount > 1;

                    // Return if there are any duplicates
                    if (hasRowDuplicate || hasColumnDuplicate || hasSubsquareDuplicate) {
                        return false;
                    }
                }
            }
        }

        // We iterated to the end and didn't find any duplicates
        return true;
    }
}
