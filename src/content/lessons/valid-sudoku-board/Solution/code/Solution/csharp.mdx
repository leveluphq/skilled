public class Solution {
    public bool IsValidSudokuBoard(int[][] board) {
        // Exit early if the board does not have a valid shape
        if (!HasValidShape(board)) {
            return false;
        }

        return HasValidCells(board);
    }

    // The function to check for a valid shape is all constant time and space O(1)
    private bool HasValidShape(int[][] board) {
        // We have an empty board, it is automatically invalid
        if (board.Length == 0 || board[0].Length == 0) {
            return false;
        }

        int rowCount = board.Length;
        int columnCount = board[0].Length;
        bool isSquare = rowCount == columnCount;

        // The board is not a square, so we know it is invalid
        if (!isSquare) {
            return false;
        }

        // For a board to be composed of sub-squares,
        // the square root of `k` must be a whole number
        bool hasSubsquares = Math.Sqrt(rowCount) % 1 == 0;
        if (!hasSubsquares) {
            return false;
        }

        // All conditions are met, so it is a valid shape
        return true;
    }

    // Helper function to init our array of hash tables
    private static Dictionary<int, int>[] ArrayOfHashTables(int size) {
        var hashTables = new Dictionary<int, int>[size];
        for (int i = 0; i < size; i++) {
            hashTables[i] = new Dictionary<int, int>();
        }
        return hashTables;
    }

    private bool HasValidCells(int[][] board) {
        // Get the dimensions of the board and sub-squares
        int sizeOfBoard = board.Length;
        int sizeOfSubsquare = (int)Math.Sqrt(sizeOfBoard);

        // Create our one-dimensional array of hash tables for rows and columns
        Dictionary<int, int>[] rows = ArrayOfHashTables(sizeOfBoard);
        Dictionary<int, int>[] columns = ArrayOfHashTables(sizeOfBoard);

        // Create a two-dimensional array of hash tables
        // where there is a separate hash table for each sub-square
        var subsquares = new Dictionary<int, int>[sizeOfSubsquare][];
        for (int i = 0; i < sizeOfSubsquare; i++) {
            subsquares[i] = ArrayOfHashTables(sizeOfSubsquare);
        }

        for (int row = 0; row < sizeOfBoard; row++) {
            for (int col = 0; col < sizeOfBoard; col++) {
                int cellValue = board[row][col];

                // If we have a value, we need to add it to our hash tables
                // and return false if a value already existed.
                if (cellValue > 0) {
                    // Since the sub-square tracker is a 2D jxj array,
                    // we need to normalize the coordinates by j
                    int subsquareRow = row / sizeOfSubsquare;
                    int subsquareCol = col / sizeOfSubsquare;
                    Dictionary<int, int> subsquare = subsquares[subsquareRow][subsquareCol];

                    // Initialize and/or increment the cell value for the corresponding row, column, sub-square
                    //
                    // rows and columns are a 1-D array where we look for the cellValue in a hash table
                    rows[row][cellValue] = rows[row].GetValueOrDefault(cellValue) + 1;
                    columns[col][cellValue] = columns[col].GetValueOrDefault(cellValue) + 1;
                    //
                    // subsquares is a 2-D matrix
                    subsquare[cellValue] = subsquare.GetValueOrDefault(cellValue) + 1;

                    // If the value is greater than 1, we have already seen this in the hash table
                    bool hasRowDuplicate = rows[row][cellValue] > 1;
                    bool hasColumnDuplicate = columns[col][cellValue] > 1;
                    bool hasSubsquareDuplicate = subsquare[cellValue] > 1;

                    // Return if there are any duplicates
                    if (hasRowDuplicate || hasColumnDuplicate || hasSubsquareDuplicate) {
                        return false;
                    }
                }
            }
        }

        // We iterated to the end and didn't find any duplicates
        return true;
    }
}
