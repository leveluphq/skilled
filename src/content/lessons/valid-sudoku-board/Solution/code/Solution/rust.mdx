```rust
use std::collections::HashMap;

impl Solution {
    pub fn is_valid_sudoku_board(board: Vec<Vec<i32>>) -> bool {
        // Exit early if the board does not have a valid shape
        if !has_valid_shape(&board) {
            return false;
        }

        has_valid_cells(&board)
    }
}

// The function to check for a valid shape is all constant time and space O(1)
fn has_valid_shape(board: &[Vec<i32>]) -> bool {
    // We have an empty board, it is automatically invalid
    if board.is_empty() || board[0].is_empty() {
        return false;
    }

    let row_count = board.len();
    let column_count = board[0].len();
    let is_square = row_count == column_count;

    // The board is not a square, so we know it is invalid
    if !is_square {
        return false;
    }

    // For a board to be composed of sub-squares,
    // the square root of `k` must be a whole number
    let has_subsquares = (row_count as f64).sqrt().fract() == 0.0;
    if !has_subsquares {
        return false;
    }

    // All conditions are met, so it is a valid shape
    true
}

fn has_valid_cells(board: &[Vec<i32>]) -> bool {
    // Get the dimensions of the board and sub-squares
    let size_of_board = board.len();
    let size_of_subsquare = (size_of_board as f64).sqrt() as usize;

    // Create our one-dimensional lists of hash tables for rows and columns
    let mut rows: Vec<HashMap<i32, i32>> = vec![HashMap::new(); size_of_board];
    let mut columns: Vec<HashMap<i32, i32>> = vec![HashMap::new(); size_of_board];

    // Create a two-dimensional list of hash tables
    // where there is a separate hash table for each sub-square
    let mut subsquares: Vec<Vec<HashMap<i32, i32>>> =
        vec![vec![HashMap::new(); size_of_subsquare]; size_of_subsquare];

    for row in 0..size_of_board {
        for col in 0..size_of_board {
            let cell_value = board[row][col];

            // If we have a value, we need to add it to our hash tables
            // and return false if a value already existed.
            if cell_value > 0 {
                // Since the sub-square tracker is a 2D jxj list,
                // we need to normalize the coordinates by j
                let subsquare_row = row / size_of_subsquare;
                let subsquare_col = col / size_of_subsquare;

                // Initialize and/or increment the cell value for the corresponding row, column, sub-square
                //
                // rows and columns are a 1-D list where we look for the cell_value in a hash table
                *rows[row].entry(cell_value).or_insert(0) += 1;
                *columns[col].entry(cell_value).or_insert(0) += 1;
                //
                // subsquares is a 2-D matrix
                *subsquares[subsquare_row][subsquare_col]
                    .entry(cell_value)
                    .or_insert(0) += 1;

                // If the value is greater than 1, we have already seen this in the hash table
                let has_row_duplicate = rows[row][&cell_value] > 1;
                let has_column_duplicate = columns[col][&cell_value] > 1;
                let has_subsquare_duplicate =
                    subsquares[subsquare_row][subsquare_col][&cell_value] > 1;

                // Return if there are any duplicates
                if has_row_duplicate || has_column_duplicate || has_subsquare_duplicate {
                    return false;
                }
            }
        }
    }

    // We iterated to the end and didn't find any duplicates
    true
}
```
