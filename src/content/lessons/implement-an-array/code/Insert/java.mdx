    if (index > length - 1 || index < 0) {
        return null;
    }

    // Create our new length
    length++;

    // We iterate from the back since this is our new empty item
    for (int i = length - 1; i > index; i--) {
        data.put(i, data.get(i - 1));
    }

    // Add the new item
    data.put(index, item);

    // Return the array
    return data;
}
