func firstBadVersion(n int) int {
	lowerBound := 1 // Starts at v1
	upperBound := n

	for lowerBound < upperBound {
		midpoint := lowerBound + (upperBound-lowerBound)/2

		if isBadVersion(midpoint) {
			upperBound = midpoint
		} else {
			lowerBound = midpoint + 1
		}
	}

	return lowerBound
}
