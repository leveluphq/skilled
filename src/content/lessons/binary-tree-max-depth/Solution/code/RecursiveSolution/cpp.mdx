```cpp
class Solution {
public:
    int findMaxDepth(BinaryNode* root) {
        if (root == nullptr) {
            return -1;
        }

        int leftDepth = findMaxDepth(root->left);
        int rightDepth = findMaxDepth(root->right);

        return max(leftDepth, rightDepth) + 1;
    }
};
```
