class Solution {
    fun groupPermutationsOfPalindromes(strings: Array<String>): List<List<String>> {
        val permutationsOfPalindromes = mutableMapOf<String, MutableList<String>>()

        // Iterate through all O(n) strings in our array
        for (string in strings) {
            // Check if each string is a palindrome O(c)
            if (isPermutationOfPalindrome(string)) {
                // Sort our string's order O(c log c)
                // Permutations will sort to the same string
                val sortedString = string.toCharArray().sorted().joinToString("")

                // Initialize the key to a list if we haven't seen
                // any of the string's permutations yet, then add the string
                permutationsOfPalindromes.getOrPut(sortedString) { mutableListOf() }.add(string)
            }
        }

        // The values of the hash table are lists of strings,
        // which is the form of the result we want
        return permutationsOfPalindromes.values.toList()
    }

    // We use the modulo operator to determine if a number is odd
    private fun isOdd(num: Int) = num % 2 != 0

    // All O(c) operations
    private fun isPermutationOfPalindrome(string: String): Boolean {
        // Use a hash table to count the occurrence of each character in a string
        // These are sequential O(c) operations
        val characterCountTable = mutableMapOf<Char, Int>()

        for (char in string) {
            characterCountTable[char] = (characterCountTable[char] ?: 0) + 1
        }

        // Sum the number of characters that contain an odd count
        val oddCharacterCount = characterCountTable.values.count { numCharacters -> isOdd(numCharacters) }

        // If only one character has an odd count, this function returns true
        return oddCharacterCount <= 1
    }
}
