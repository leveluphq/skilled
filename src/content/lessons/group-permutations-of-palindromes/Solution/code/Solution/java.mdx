class Solution {
    public String[][] groupPermutationsOfPalindromes(String[] strings) {
        Map<String, List<String>> permutationsOfPalindromes = new HashMap<>();

        // Iterate through all O(n) strings in our array
        for (String string : strings) {
            // Check if each string is a palindrome O(c)
            if (isPermutationOfPalindrome(string)) {
                // Sort our string's order O(c log c)
                // Permutations will sort to the same string
                char[] characters = string.toCharArray();
                Arrays.sort(characters);
                String sortedString = new String(characters);

                // Initialize the key to a list if we
                // haven't seen any of the string's permutations yet
                permutationsOfPalindromes.computeIfAbsent(sortedString, key -> new ArrayList<>()).add(string);
            }
        }

        // Turn the values of the hash table into an array of arrays
        // Since the values are lists of strings, this is the form of the result we want
        return permutationsOfPalindromes.values().stream()
            .map(group -> group.toArray(new String[0]))
            .toArray(String[][]::new);
    }

    // We use the modulo operator to determine if a number is odd
    private boolean isOdd(int num) {
        return num % 2 != 0;
    }

    // All O(c) operations
    private boolean isPermutationOfPalindrome(String string) {
        // Use a hash table to count the occurrence of each character in a string
        // These are sequential O(c) operations
        Map<Character, Integer> characterCountTable = new HashMap<>();

        for (char character : string.toCharArray()) {
            characterCountTable.merge(character, 1, Integer::sum);
        }

        // Sum the number of characters that contain an odd count
        int oddCharacterCount = 0;
        for (int numCharacters : characterCountTable.values()) {
            if (isOdd(numCharacters)) {
                oddCharacterCount++;
            }
        }

        // If only one character has an odd count, this function returns true
        return oddCharacterCount <= 1;
    }
}
