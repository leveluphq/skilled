class Solution {
    func groupPermutationsOfPalindromes(_ strings: [String]) -> [[String]] {
        var permutationsOfPalindromes: [String: [String]] = [:]

        // Iterate through all O(n) strings in our array
        for string in strings {
            // Check if each string is a palindrome O(c)
            if isPermutationOfPalindrome(string) {
                // Sort our string's order O(c log c)
                // Permutations will sort to the same string
                let sortedString = String(string.sorted())

                // Add the string to its group, starting with an empty
                // array if we haven't seen any of the string's permutations yet
                permutationsOfPalindromes[sortedString, default: []].append(string)
            }
        }

        // This will return all the values of the hash table as an array
        // Since the values are also arrays, this is the form of the result we want
        return Array(permutationsOfPalindromes.values)
    }

    // We use the remainder operator to determine if a number is odd
    private func isOdd(_ num: Int) -> Bool {
        return num % 2 != 0
    }

    // All O(c) operations
    private func isPermutationOfPalindrome(_ string: String) -> Bool {
        // Use a hash table to count the occurrence of each character in a string
        // These are sequential O(c) operations
        var characterCountTable: [Character: Int] = [:]

        for char in string {
            characterCountTable[char, default: 0] += 1
        }

        // Sum the number of characters that contain an odd count
        var oddCharacterCount = 0
        for numCharacters in characterCountTable.values where isOdd(numCharacters) {
            oddCharacterCount += 1
        }

        // If only one character has an odd count, this function returns true
        return oddCharacterCount <= 1
    }
}
