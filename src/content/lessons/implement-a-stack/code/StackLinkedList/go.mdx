```go
type Node struct {
	Data int
	Prev *Node
}

type Stack struct {
	last *Node
}

func NewStack() *Stack {
	return &Stack{last: nil}
}

func (s *Stack) Peek() *Node {
	return s.last
}

func (s *Stack) Push(item int) *Node {
	previousLastItem := s.last

	s.last = &Node{Data: item}
	s.last.Prev = previousLastItem

	return s.last
}

func (s *Stack) Pop() *Node {
	removedItem := s.last

	// Update `last` as long as we have items left in the stack
	if removedItem != nil {
		s.last = removedItem.Prev
	}

	return removedItem
}
```
