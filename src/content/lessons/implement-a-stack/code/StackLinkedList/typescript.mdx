```typescript
class StackNode {
  data: any;
  prev: StackNode | null;

  constructor(data: any) {
    this.data = data;
    this.prev = null;
  }
}

class Stack {
  last: StackNode | null;

  constructor() {
    this.last = null;
  }

  peek(): StackNode | null {
    return this.last;
  }

  push(item: any): StackNode | null {
    const previousLastItem = this.last;

    this.last = new StackNode(item);
    this.last.prev = previousLastItem;

    return this.last;
  }

  pop(): StackNode | null {
    const removedItem = this.last;

    // Update `last` as long as we have items left in the stack
    if (removedItem) {
      this.last = removedItem.prev;
    }

    return removedItem;
  }
}
```
