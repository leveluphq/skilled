fun bruteForceFindEmail(emails: Array<String>): String {
    var maxCount = 0
    var email = ""

    // Nested for loops - immediately think O(n^2)
    for (i in emails.indices) {
        val currentEmail = emails[i]
        var currentCount = 0

        for (j in emails.indices) {
            if (emails[j] == currentEmail) {
                currentCount++
            }
        }

        val isNewMax = currentCount > maxCount
        maxCount = if (isNewMax) currentCount else maxCount
        email = if (isNewMax) currentEmail else email
    }

    return email
}
