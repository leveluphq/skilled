string bruteForceFindEmail(vector<string>& emails) {
    int maxCount = 0;
    string email = "";

    // Nested for loops - immediately think O(n^2)
    for (int i = 0; i < (int) emails.size(); i++) {
        const string& currentEmail = emails[i];
        int currentCount = 0;

        for (int j = 0; j < (int) emails.size(); j++) {
            if (emails[j] == currentEmail) {
                currentCount++;
            }
        }

        bool isNewMax = currentCount > maxCount;
        maxCount = isNewMax ? currentCount : maxCount;
        email = isNewMax ? currentEmail : email;
    }

    return email;
}
