enqueue(item) {
  const newLastNode = new Node(item);
  const isEmptyList = !this.first && !this.last;

  if (isEmptyList) {
    this.first = newLastNode;
    this.last = newLastNode;
  } else {
    this.last.prev = newLastNode;
    this.last = newLastNode;
  }

  return newLastNode;
}
