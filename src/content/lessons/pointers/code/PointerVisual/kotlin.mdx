```kotlin
data class Greeting(var hi: String, var friendly: Boolean)

var pointer: Greeting? = Greeting(hi = "world", friendly = true)
// The object Greeting(hi=world, friendly=true) is created in memory
// The pointer variable references/points to the object
// pointer ----> Greeting(hi=world, friendly=true)

println(pointer?.hi)
// pointer accesses the object from memory and requests the value stored for `hi`
// pointer ----> Greeting(hi=world, friendly=true) ----> hi
//                                                 <---- "world"

pointer?.hi = "friends"
// pointer reassigns the value for `hi`
// pointer ----> Greeting(hi=world, friendly=true) ----> hi = "friends"

pointer = null
// pointer changes its reference from the object to null
// The object still exists in memory but is just no longer referenced to by the variable
//
// pointer           Greeting(hi=friends, friendly=true)
//       |
//       |
//       |--> null
```
