const tests = [
  {
    input: [{ method: 'push', value: 1 }, { method: 'push', value: 42 }],
    expected: { min: 1, max: 42 },
    description: 'simple min max',
  },
  {
    input: [{ method: 'push', value: 1 }, { method: 'push', value: 42 }, { method: 'push', value: -10 }, { method: 'push', value: 100 }],
    expected: { min: -10, max: 100 },
    description: 'change min/max',
  },
  {
    input: [{ method: 'push', value: 1 }, { method: 'push', value: 42 }, { method: 'push', value: -10 }, { method: 'push', value: 100 }, { method: 'pop' }, { method: 'pop' }, { method: 'pop' }],
    expected: { min: 1, max: 1 },
    description: 'push and pop',
  },
];

const validate = ({ input, expected, description }) => {
  const stack = new MaxMinStack();

  input.forEach(item => stack[item.method](item.value));
  const max = stack.getStackMax();
  const min = stack.getStackMin();

  const isCorrectMax = expected.max === max;
  const isCorrectMin = expected.min === min;

  const passed = isCorrectMax && isCorrectMin;

  return { passed, description, expected, userAnswer: stack };
};
