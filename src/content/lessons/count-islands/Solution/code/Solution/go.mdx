const (
	land  = '#'
	water = '~'
)

func countIslands(grid [][]byte) int {
	if len(grid) == 0 || len(grid[0]) == 0 {
		return 0
	}

	numRows := len(grid)
	numCols := len(grid[0])
	islandCount := 0

	for row := 0; row < numRows; row++ {
		for col := 0; col < numCols; col++ {
			// We found another uncounted island
			if grid[row][col] == land {
				islandCount++
				// Mark all the pieces of land for the island as viewed
				markLandAsViewedDFS(grid, row, col)
			}
		}
	}

	return islandCount
}

func markLandAsViewedDFS(grid [][]byte, row int, col int) {
	numRows := len(grid)
	numCols := len(grid[0])

	// Stopping condition:
	// 1. Outside of the map
	// 2. Reached an adjacent water node
	isOutsideMap := row < 0 || col < 0 || row >= numRows || col >= numCols
	if isOutsideMap || grid[row][col] == water {
		return
	}

	// Recursive case:
	// 1. Mark the node as viewed by flipping it to water
	// 2. Check the nodes adjacent to the current
	grid[row][col] = water
	markLandAsViewedDFS(grid, row-1, col)
	markLandAsViewedDFS(grid, row+1, col)
	markLandAsViewedDFS(grid, row, col-1)
	markLandAsViewedDFS(grid, row, col+1)
}
