const val LAND = '#'
const val WATER = '~'

class Solution {
    fun countIslands(grid: Array<CharArray>): Int {
        if (grid.isEmpty() || grid[0].isEmpty()) {
            return 0
        }

        val numRows = grid.size
        val numCols = grid[0].size
        var islandCount = 0

        for (row in 0 until numRows) {
            for (col in 0 until numCols) {
                // We found another uncounted island
                if (grid[row][col] == LAND) {
                    islandCount = islandCount + 1
                    // Mark all the pieces of land for the island as viewed
                    markLandAsViewedDFS(grid, row, col)
                }
            }
        }

        return islandCount
    }

    private fun markLandAsViewedDFS(grid: Array<CharArray>, row: Int, col: Int) {
        val numRows = grid.size
        val numCols = grid[0].size

        // Stopping condition:
        // 1. Outside of the map
        // 2. Reached an adjacent water node
        val isOutsideMap = row < 0 || col < 0 || row >= numRows || col >= numCols
        if (isOutsideMap || grid[row][col] == WATER) {
            return
        }

        // Recursive case:
        // 1. Mark the node as viewed by flipping it to water
        // 2. Check the nodes adjacent to the current
        grid[row][col] = WATER
        markLandAsViewedDFS(grid, row - 1, col)
        markLandAsViewedDFS(grid, row + 1, col)
        markLandAsViewedDFS(grid, row, col - 1)
        markLandAsViewedDFS(grid, row, col + 1)
    }
}
