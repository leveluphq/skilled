const LAND: char = '#';
const WATER: char = '~';

impl Solution {
    pub fn count_islands(mut grid: Vec<Vec<char>>) -> i32 {
        if grid.is_empty() || grid[0].is_empty() {
            return 0;
        }

        let num_rows = grid.len();
        let num_cols = grid[0].len();
        let mut island_count = 0;

        for row in 0..num_rows {
            for col in 0..num_cols {
                // We found another uncounted island
                if grid[row][col] == LAND {
                    island_count += 1;
                    // Mark all the pieces of land for the island as viewed
                    mark_land_as_viewed_dfs(&mut grid, row as i32, col as i32);
                }
            }
        }

        island_count
    }
}

fn mark_land_as_viewed_dfs(grid: &mut [Vec<char>], row: i32, col: i32) {
    let num_rows = grid.len() as i32;
    let num_cols = grid[0].len() as i32;

    // Stopping condition:
    // 1. Outside of the map
    // 2. Reached an adjacent water node
    let is_outside_map = row < 0 || col < 0 || row >= num_rows || col >= num_cols;
    if is_outside_map || grid[row as usize][col as usize] == WATER {
        return;
    }

    // Recursive case:
    // 1. Mark the node as viewed by flipping it to water
    // 2. Check the nodes adjacent to the current
    grid[row as usize][col as usize] = WATER;
    mark_land_as_viewed_dfs(grid, row - 1, col);
    mark_land_as_viewed_dfs(grid, row + 1, col);
    mark_land_as_viewed_dfs(grid, row, col - 1);
    mark_land_as_viewed_dfs(grid, row, col + 1);
}
