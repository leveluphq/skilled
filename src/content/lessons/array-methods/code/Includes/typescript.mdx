```javascript
const array = ['Jeni', 'Bob', 'Carol'];

const isFriend = (friends: string[], name: string): boolean => {
  for (let i = 0; i < friends.length; i++) {
    const friend = friends[i];
    if (friend === name) {
      return true;
    }
  }
  return false;
}

console.log(isFriend(array, 'Carol')); // true
```
