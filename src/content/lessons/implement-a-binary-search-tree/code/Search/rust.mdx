fn search(&self, data: i32) -> Option<&BinaryNode> {
    let mut current_node = self.root.as_deref();

    while let Some(node) = current_node {
        if data < node.data {
            current_node = node.left.as_deref();
        } else if data > node.data {
            current_node = node.right.as_deref();
        } else {
            return Some(node);
        }
    }

    None
}
