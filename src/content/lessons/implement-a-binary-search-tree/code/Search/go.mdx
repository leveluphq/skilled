func (t *BinarySearchTree) Search(data int) *BinaryNode {
	if t.root == nil {
		return nil
	}

	currentNode := t.root

	for currentNode != nil {
		if data < currentNode.Data {
			currentNode = currentNode.Left
		} else if data > currentNode.Data {
			currentNode = currentNode.Right
		} else {
			return currentNode
		}
	}

	return nil
}
