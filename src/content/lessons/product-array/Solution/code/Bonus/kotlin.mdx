class Solution {
    fun buildProductArrayWithDivision(nums: LongArray): LongArray {
        // Zeroes are now an edge case because we can't divide by them
        var numZeros = 0
        // Init product to 1
        var product = 1L

        nums.forEach { element ->
            if (element == 0L) {
                numZeros++
            } else {
                product = product * element
            }
        }

        for (i in nums.indices) {
            if (numZeros == 0) {
                // If no zeroes, don't worry about edge cases
                nums[i] = product / nums[i]
            } else if (numZeros == 1 && nums[i] == 0L) {
                // Handle the 1 zero where it will have the only non-zero product
                nums[i] = product
            } else {
                // If we have more than one 0, they're all 0
                nums[i] = 0
            }
        }

        return nums
    }
}
