impl Solution {
    pub fn build_product_array(nums: Vec<i64>) -> Vec<i64> {
        let mut results = vec![1; nums.len()];
        let mut left_running_product = 1;

        // First use our results list as the running right product list.
        // We fill from the right side.
        for i in 1..nums.len() {
            let end_idx = nums.len() - 1 - i;
            results[end_idx] = results[end_idx + 1] * nums[end_idx + 1];
        }

        // Refill the results with the product at each point
        for i in 0..nums.len() {
            let product_except_self = left_running_product * results[i];

            // We use results[i] above so now it can be replaced
            results[i] = product_except_self;

            // Now that we have used our left product,
            // we update it to include the current element.
            left_running_product *= nums[i];
        }

        results
    }
}
