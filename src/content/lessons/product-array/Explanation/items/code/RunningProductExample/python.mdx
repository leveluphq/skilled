right_running_product = [|2 * 3 * 4 * 5 * 1|, |3 * 4 * 5 * 1|, |4 * 5 * 1|, |5 * 1|, 1]

left_running_product = [1, 1, 2, 6, 24]

results = [

    left_running_product[1] * right_running_product[1],
    left_running_product[2] * right_running_product[2],
    left_running_product[3] * right_running_product[3],
    left_running_product[len(input) - 1] * right_running_product[len(input) - 1]
]
