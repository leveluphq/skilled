func runningProductSolution(input []int64) []int64 {
	results := []int64{}
	length := len(input)
	// Initialize our lookup arrays and fill them with 1's
	leftRunningProduct := slices.Repeat([]int64{1}, length)
	rightRunningProduct := slices.Repeat([]int64{1}, length)

	// Begin creating the product arrays 1 element inward at i = 1
	// since the first product in each array is supposed to be 1.
	for i := 1; i < length; i++ {
		leftRunningProduct[i] = leftRunningProduct[i-1] * input[i-1]

		// We will fill our right product array from the end
		endIdx := length - 1 - i
		rightRunningProduct[endIdx] = rightRunningProduct[endIdx+1] * input[endIdx+1]
	}

	// We can now find the product of all the items other than the current index
	// by multiplying the product of the elements to its left and right together.
	for i := range length {
		leftProduct := leftRunningProduct[i]
		rightProduct := rightRunningProduct[i]
		productExceptSelf := leftProduct * rightProduct

		results = append(results, productExceptSelf)
	}

	return results
}
