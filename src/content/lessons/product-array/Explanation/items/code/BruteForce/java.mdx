public long[] bruteForceAnswer(long[] nums) {
    long[] results = new long[nums.length];

    for (int i = 0; i < nums.length; i++) {
        long product = 1;

        for (int j = 0; j < nums.length; j++) {
            if (i != j) {
                product = product * nums[j];
            }
        }

        results[i] = product;
    }

    return results;
}
