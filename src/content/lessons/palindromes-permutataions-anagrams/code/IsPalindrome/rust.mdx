// Remove non-alphanumeric characters
// (Rust's standard library has no regex, so we filter the characters)
fn remove_non_alphanumeric(string: &str) -> String {
    string
        .chars()
        .filter(|character| character.is_ascii_alphanumeric())
        .collect()
}

fn is_palindrome(string: &str) -> bool {
    // For simplicity, I'm doing the transforming here at an assumed O(n) cost
    // You could also not transform it and consider this when you compare characters
    // (A Rust string can't be indexed by position, so we collect its characters in a Vec)
    let formatted_string: Vec<char> = remove_non_alphanumeric(string)
        .to_lowercase()
        .chars()
        .collect();
    let length = formatted_string.len();

    // Walk from the beginning and end to the middle and compare each character
    for i in 0..length / 2 {
        // If the characters are ever not equal, it's not a palindrome
        if formatted_string[i] != formatted_string[length - 1 - i] {
            return false;
        }
    }

    true
}
