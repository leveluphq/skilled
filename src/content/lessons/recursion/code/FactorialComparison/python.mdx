# Ex. factorial of 4: 4! = 4 * 3 * 2 * 1 = 24

def iterative_factorial(number):
    factorial = 1

    for i in range(1, number + 1):
        factorial = factorial * i

    return factorial

# Same as:
# 4! = 4 * 3 * 2 * 1
#    = 4 * 3!
#    = 4 * 3 * 2!
#    = 4 * 3 * 2 * 1!
#    = 4 * 3 * 2 * 1

def recursive_factorial(number):
    # Stop calling when the value is 1 or less
    if number <= 1:
        return 1

    return number * recursive_factorial(number - 1)
