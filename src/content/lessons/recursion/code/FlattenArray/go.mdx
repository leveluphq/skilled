func recursiveFlatten(array []any) []int {
	flattenedArray := []int{}

	for _, item := range array {
		// if the item is a slice, flatten it with a recursive call
		if nested, ok := item.([]any); ok {
			flattenedArray = append(flattenedArray, recursiveFlatten(nested)...)
		} else {
			flattenedArray = append(flattenedArray, item.(int))
		}
	}

	return flattenedArray
}

	output := []int{}
	toBeFlattened := []any{array}
