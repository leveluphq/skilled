// Stands in for Infinity: no valid coin combination for an amount (yet)
const val INFINITY = Int.MAX_VALUE

class Solution {
    fun minCoinsForChange(denominations: IntArray, target: Int): Int {
        /*
          Init all values to INFINITY which indicates there isn't a valid coin combination for an amount (yet)
          This is our dp / memo array that we're using to memoize the solution to subproblems
        */
        val minCoinsForAmount = IntArray(target + 1) { INFINITY }

        // Bottom-up, start with 0 which is always 0
        minCoinsForAmount[0] = 0

        /*
          Begin at the smallest amount and build a running table of
          the smallest values required to reach EVERY amount up to the target
        */
        for (amount in minCoinsForAmount.indices) {
            // Try each denomination to reach the current amount
            for (coin in denominations) {
                // The coin must fit, and (amount - coin) must be reachable:
                // INFINITY + 1 would overflow an Int
                if (coin <= amount && minCoinsForAmount[amount - coin] != INFINITY) {
                    // The current minimum (subproblem solution) found at this amount.
                    val existingMinCoinForAmount = minCoinsForAmount[amount]
                    /*
                      Subtract (amount - coin) to find the min coins to reach this location from a prior value.
                      Always increment by +1 because we are trying one additional coin.
                    */
                    val minAmountWithCurrentCoin = minCoinsForAmount[amount - coin] + 1

                    /*
                      Memoize the minimum coins to reach the current amount.
                      This is either a minimum we previously calculated or a new minimum we found with the current coin.
                      If there isn't a valid combination, this just remains INFINITY.
                    */
                    minCoinsForAmount[amount] = minOf(existingMinCoinForAmount, minAmountWithCurrentCoin)
                }
            }
        }

        /*
          If we didn't find a combination that matched our target (it's still INFINITY), return -1.
          Otherwise, the target does exist for that amount and we return it.
        */
        return if (minCoinsForAmount[target] == INFINITY) -1 else minCoinsForAmount[target]
    }
}
