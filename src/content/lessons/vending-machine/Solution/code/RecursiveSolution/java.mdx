class Solution {
    public int minCoinsForChange(int[] denominations, int target) {
        if (target < 1) {
            return 0;
        }

        /*
          Init all values to 0 which indicates no coins yet match the target
          (Java fills a new int array with 0s)
          This is our dp / memo array that we're using to memoize the solution to subproblems
        */
        int[] minCoinsForAmount = new int[target + 1];

        return minCoinsForChangeMemo(denominations, target, minCoinsForAmount);
    }

    private int minCoinsForChangeMemo(int[] denominations, int remaining, int[] minCoinsForAmount) {
        /*
          Base case 1: Invalid remaining coins
          We use -1 to indicate it's invalid
        */
        if (remaining < 0) {
            return -1;
        }

        // Reached zero, valid case
        if (remaining == 0) {
            return 0;
        }

        // Check if we have a memoized (already calculated) value for this amount remaining amount
        if (minCoinsForAmount[remaining] != 0) {
            return minCoinsForAmount[remaining];
        }

        // Java ints have no Infinity, so the largest int stands in for it
        int minStepsForRemaining = Integer.MAX_VALUE;

        // Subtract each coin from the current remaining amount (fork paths from this amount)
        for (int coin : denominations) {
            int stepsForCoin = minCoinsForChangeMemo(denominations, remaining - coin, minCoinsForAmount);

            /*
              If stepsForCoin >= 0, it's possible to make change with this denomination,
              and the steps for this denomination is less than the current min,
              we increment the steps to account for it as the new minimum.

              stepsForCoin = -1 means it's invalid which is why we check for >= 0
            */
            if (stepsForCoin >= 0 && stepsForCoin < minStepsForRemaining) {
                minStepsForRemaining = stepsForCoin + 1;
            }
        }

        /*
          If we're able to find a minStepsForRemaining for the current subproblem, we memoize it.
          If minStepsForRemaining is still Integer.MAX_VALUE, it's not valid and we use our invalid placeholder -1.
        */
        minCoinsForAmount[remaining] = (minStepsForRemaining == Integer.MAX_VALUE) ? -1 : minStepsForRemaining;

        return minCoinsForAmount[remaining];
    }
}
