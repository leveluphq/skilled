def get_max_profit(prices):
    if len(prices) < 2:
        return 0

    max_profit = 0
    min_price = prices[0]

    for price in prices:
        profit_at_current_day = price - min_price
        max_profit = max(max_profit, profit_at_current_day)
        min_price = min(price, min_price)

    return max_profit
